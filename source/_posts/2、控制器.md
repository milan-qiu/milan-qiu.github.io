<p>一、控制器的目录为：app / Http / Controllers / XxxController.php</p>
<div class="cnblogs_code">
<pre>控制器命名规则：XxxController.php</pre>
</div>
<p>&nbsp;</p>
<p>二、控制器页面基本内容：</p>
<div class="cnblogs_code">
<pre>&lt;?<span style="color: #000000;">php

</span><span style="color: #0000ff;">namespace</span> App\Http\Controllers;  <span style="color: #008000;">//</span><span style="color: #008000;">命名空间，指的是本控制器的位置</span>

<span style="color: #0000ff;">class</span> XxxController extends BaseController  <span style="color: #008000;">//</span><span style="color: #008000;">这里指Xxx控制器 继承 基本控制器，简称基类</span>
<span style="color: #000000;">
{
    </span><span style="color: #0000ff;">public</span> function index () {  <span style="color: #008000;">//</span><span style="color: #008000;">index触发器</span>
        <span style="color: #0000ff;">return</span> <span style="color: #800000;">'</span><span style="color: #800000;">hello</span><span style="color: #800000;">'</span><span style="color: #000000;">;
    }
}</span></pre>
</div>
<p>ps：路由快捷创建</p>
<div class="cnblogs_code">
<pre>php artisan make:controller XxxController</pre>
</div>
<p>&nbsp;</p>
<p>&nbsp;</p>
<p>三、在路由中关联&nbsp; 控制器和触发器</p>
<div class="cnblogs_code">
<pre>Route::<span style="color: #0000ff;">get</span>(<span style="color: #800000;">'</span><span style="color: #800000;">/</span><span style="color: #800000;">'</span>,[<span style="color: #800000;">XxxController::class , 'index</span><span style="color: #800000;">']</span><span style="color: #000000;">);<br />
</span><span style="color: #008000;">//</span><span style="color: #008000;">或<br />
</span><span style="color: #008000;">//</span><span style="color: #008000;"> Route::get('/',['user' =&gt; 'XxxController@index']);
</span><span style="color: #008000;">//</span><span style="color: #008000;">这里除了get还可以是post/any等等<br /><br />在laravel8中要记得引入使用到的路由<br /></span></pre>
<pre>use App\Http\Controllers\XxxController;</pre>
</div>
<p>&nbsp;</p>
<p><span style="text-decoration: line-through;">路由别名</span></p>
<div class="cnblogs_code">
<pre><span style="color: #008000;">//</span><span style="color: #008000;">路由写</span>
Route::<span style="color: #0000ff;">get</span>(<span style="color: #800000;">'</span><span style="color: #800000;">/</span><span style="color: #800000;">'</span><span style="color: #000000;">,[
    </span><span style="color: #800000;">'</span><span style="color: #800000;">user</span><span style="color: #800000;">'</span> =&gt; <span style="color: #800000;">'</span><span style="color: #800000;">XxxController@index</span><span style="color: #800000;">'</span><span style="color: #000000;">,
    </span><span style="color: #800000;">'</span><span style="color: #800000;">as</span><span style="color: #800000;">'</span> =&gt; <span style="color: #800000;">'</span><span style="color: #800000;">bb</span><span style="color: #800000;">'</span><span style="color: #000000;">
]);
</span><span style="color: #008000;">//</span><span style="color: #008000;"> 控制器写</span>
<span style="color: #0000ff;">return</span> route(<span style="color: #800000;">'</span><span style="color: #800000;">bb</span><span style="color: #800000;">'</span>);<span style="color: #008000;">//</span><span style="color: #008000;">返回的是这个路由的地址，laravel7似乎报错</span></pre>
</div>
<p>&nbsp;</p>
<p>路由传参</p>
<div class="cnblogs_code">
<pre><span style="color: #008000;">//</span><span style="color: #008000;">路由传id</span>
Route::<span style="color: #0000ff;">get</span>(<span style="color: #800000;">'</span><span style="color: #800000;">/{id}</span><span style="color: #800000;">'</span> , <span style="color: #800000;">'</span><span style="color: #800000;">XxxController@index</span><span style="color: #800000;">'</span>); <span style="color: #008000;">//</span><span style="color: #008000;">将id传到index触发器

</span><span style="color: #008000;">//</span><span style="color: #008000;">控制器接收</span>
<span style="color: #0000ff;">public</span> function index($id){ <span style="color: #008000;">//</span><span style="color: #008000;">这里接收id</span>
  <span style="color: #0000ff;">return</span><span style="color: #000000;"> $id;  
}</span></pre>
</div>
<p>&nbsp;</p>
<p><strong>四、资源控制器</strong></p>
<p>创建资源控制器</p>
<div class="cnblogs_code">
<pre>php artisan make:controller UserController --resource</pre>
</div>
<p>&nbsp;</p>
<p>为资源控制器创建路由</p>
<div class="cnblogs_code">
<pre>Route::<span style="color: #0000ff;">resource</span>('users' , UserController::class);</pre>
</div>
<p>&nbsp;</p>
<p><strong>详解资源控制器的 所有函数</strong></p>
<p>index：返回所有用户数据</p>
<p>create：返回新的表单，用于提交用户信息</p>
<p>store：保存刚才表单地信息</p>
<p>show：返回单独一个用户信息</p>
<p>edit：修改用户信息的表单</p>
<p>update：保存刚才修改的表单信息</p>
<p>destroy：删除某一个用户</p>
<p>&nbsp;</p>
