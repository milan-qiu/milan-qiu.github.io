<p>用正则判断接收到的参数</p>
<div class="cnblogs_code">
<pre>Route::<span style="color: #0000ff;">get</span>(<span style="color: #800000;">'</span><span style="color: #800000;">user/{name?}</span><span style="color: #800000;">'</span> , function($name =<span style="color: #000000;"> zhangsan){

</div>

<pre>Route::<span style="color: #0000ff;">get</span>(<span style="color: #800000;">'</span><span style="color: #800000;">user/{id}/{name?}</span><span style="color: #800000;">'</span><span style="color: #000000;"> , function($id,$name){

</span><span style="color: #008000;">//</span><span style="color: #008000;">id必须数字，name必须字母，否则报错</span></pre>
