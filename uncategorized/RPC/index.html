server 端

func main() {
    // 实例化一个 server
    _ = rpc.RegisterName("HelloService", &HelleService{})
    http.HandleFunc("/wahaha",func(w http.ResponseWriter, r *http.Request) {
        var conn io.ReadWriteCloser = struct {
            io.Writer
            io.ReadCloser
        }{
            ReadCloser: r.Body,
            Writer: w,
        }
        rpc.ServeRequest(jsonrpc.NewServerCodec(conn))
    })
